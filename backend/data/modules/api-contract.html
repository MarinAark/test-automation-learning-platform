import pytest
from pact import Consumer, Provider


@pytest.fixture(scope="session")
def pact():
    pact = Consumer("UserWeb").has_pact_with(
        Provider("UserService"), port=1234
    )
    pact.start_service()
    yield pact
    pact.stop_service()
