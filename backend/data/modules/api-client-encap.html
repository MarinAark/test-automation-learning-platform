生产级 HTTP 客户端封装

封装会话管理、认证、重试、日志、超时等能力。

core/api_client.py
import json, logging
from typing import Optional
from dataclasses import dataclass, field
from urllib.parse import urljoin

import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

logger = logging.getLogger(__name__)


@dataclass
class APIConfig:
    base_url: str
    token: Optional[str] = None
    username: Optional[str] = None
    password: Optional[str] = None
    timeout: int = 30
    retry_count: int = 3
    verify_ssl: bool = True
    extra_headers: dict = field(default_factory=dict)


class APIClient:

    def __init__(self, config: APIConfig):
        self.config = config
        self.session = requests.Session()
        self.session.headers.update({
            "Content-Type": "application/json",
            "Accept": "application/json",
            **config.extra_headers,
        })

        # Token auth
        if config.token:
            self.session.headers["Authorization"] = f"Bearer {config.token}"
        elif config.username and config.password:
            self.session.auth = (config.username, config.password)

        # Retry: only on 5xx/429
        retry = Retry(
            total=config.retry_count, backoff_factor=0.5,
            status_forcelist=[429, 500, 502, 503, 504],
            allowed_methods=["GET", "POST", "PUT", "DELETE"],
        )
        adapter = HTTPAdapter(max_retries=retry)
        self.session.mount("https://", adapter)
        self.session.mount("http://", adapter)

    def _request(self, method, path, **kwargs):
        url = urljoin(self.config.base_url, path)
        timeout = kwargs.pop("timeout", self.config.timeout)
        logger.info(f"-> {method} {url}")
        resp = self.session.request(method, url, timeout=timeout,
                                     verify=self.config.verify_ssl, **kwargs)
        logger.info(f"<- {resp.status_code} [{len(resp.content)}b]")
        return resp

    def get(self, p, params=None, **kw):
        return self._request("GET", p, params=params, **kw)
    def post(self, p, data=None, **kw):
        return self._request("POST", p, json=data, **kw)
    def put(self, p, data=None, **kw):
        return self._request("PUT", p, json=data, **kw)
    def delete(self, p, **kw):
        return self._request("DELETE", p, **kw)
    def close(self):
        self.session.close()

实战场景：重试策略深度解析

接口测试中，重试不是"无脑加 3 次"就解决了。不同的故障模式需要不同的重试策略。

精细化重试控制
from urllib3.util.retry import Retry

# 错误配置：所有异常重试（包括 4xx）
bad = Retry(total=5, status_forcelist=[400, 401, 500])
# 400/401 是客户端错误，重试只会浪费时间和资源

# 正确配置：仅重试可恢复的错误
good = Retry(
    total=3,
    backoff_factor=0.5,         # 指数退避: 0.5s, 1s, 2s
    status_forcelist=[429, 500, 502, 503, 504],
    allowed_methods=["GET", "HEAD", "OPTIONS"],  # POST 慎重重试
    raise_on_status=False,      # 不自动抛异常，由上层判断
)

# 连接级别重试 vs 读取超时
from requests.adapters import HTTPAdapter
adapter = HTTPAdapter(
    max_retries=Retry(
        total=2,
        connect=1,  # 连接失败只重试 1 次
        read=2,     # 读取超时重试 2 次
        backoff_factor=0.3,
    ),
    pool_connections=10,
    pool_maxsize=20,
)
💡 生产级重试最佳实践
  • 不要重试 4xx：400/401/403/404 是客户端错误，重试不会改变结果。仅重试 5xx 和 429（限流）。
  • POST/PUT/DELETE 慎重重试：非幂等请求重试可能导致重复创建资源。必须重试时确保后端实现了幂等性（如 idempotency key）。
  • 指数退避 (Exponential Backoff)：避免"惊群效应" — 如果 10000 个测试同时重试且间隔相同，服务端瞬间雪崩。
  • 熔断器模式：重试超过阈值后暂停一段时间，给下游恢复的机会。requests 本身不支持，需引入 tenacity 或自建。
⚠️ 避坑指南：连接池耗尽与 Session 泄漏
  • 连接池耗尽：每次测试 requests.get() 而不复用 Session 时，底层 TCP 连接处于 TIME_WAIT 状态。高并发下端口耗尽导致 Cannot assign requested address。解决方案：使用全局 Session + HTTPAdapter 限制连接数。
  • Session 忘记关闭：Session 持有连接池。在 fixture teardown 中必须调用 session.close()，否则 CI 内存持续增长导致 OOM。
  • SSL 验证全局禁用：生产级项目不应 verify=False + urllib3.disable_warnings()。应在 CI 中使用内部 CA 证书或配置 REQUESTS_CA_BUNDLE 环境变量。
  • timeout 遗漏：requests 默认 timeout=None（无限等待）。一个卡死的下游会导致整个测试套件挂起。务必设置 timeout=(connect_timeout, read_timeout) 元组。
