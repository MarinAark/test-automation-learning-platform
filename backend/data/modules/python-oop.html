类与对象

基础类定义
class TestCase:
    """测试用例基类"""

    def __init__(self, name, priority="P1"):
        self.name = name
        self.priority = priority
        self._result = None  # 私有属性

    def run(self):
        """执行测试 — 子类覆盖此方法"""
        raise NotImplementedError("子类必须实现 run()")

    @property
    def result(self):
        return self._result or "未执行"

    @result.setter
    def result(self, value):
        allowed = {"通过", "失败", "跳过"}
        if value not in allowed:
            raise ValueError(f"结果必须是 {allowed}")
        self._result = value

    def __repr__(self):
        return f"TestCase(name={self.name!r}, priority={self.priority!r})"

# 使用
tc = TestCase("登录用例", "P0")
print(tc)  # TestCase(name='登录用例', priority='P0')
tc.result = "通过"
print(tc.result)  # 通过

继承与多态

继承示例
class APITestCase(TestCase):
    """接口测试用例"""
    def __init__(self, name, endpoint, method="GET", priority="P1"):
        super().__init__(name, priority)
        self.endpoint = endpoint
        self.method = method

    def run(self):
        print(f"发送 {self.method} {self.endpoint}")
        # 实际请求逻辑...
        self.result = "通过"

class UITestCase(TestCase):
    """UI 测试用例"""
    def __init__(self, name, page_url, priority="P1"):
        super().__init__(name, priority)
        self.page_url = page_url

    def run(self):
        print(f"打开页面 {self.page_url}")
        # Playwright 操作...
        self.result = "通过"

# 多态 — 统一调用
suite = [
    APITestCase("获取用户列表", "/api/users"),
    UITestCase("登录页面验证", "/login"),
]
for case in suite:
    case.run()
    print(f"  {case.name}: {case.result}")

魔术方法

魔术方法
class Assertion:
    """可组合的断言"""
    def __init__(self, actual):
        self.actual = actual

    def __eq__(self, expected):
        assert self.actual == expected, \
            f"期望 {expected!r}，实际 {self.actual!r}"
        return True

    def __gt__(self, other):
        assert self.actual > other, \
            f"期望 > {other}，实际 {self.actual}"
        return True

    def __contains__(self, item):
        assert item in self.actual, \
            f"期望包含 {item!r}，实际 {self.actual!r}"
        return True

# 使用
resp = {"status": "ok", "code": 200}
Assertion(resp["status"]) == "ok"
Assertion(resp["code"]) > 100
# Assertion(resp["status"]) == "error"  # 会抛出 AssertionError

实战场景：用组合替代深层继承

测试框架中常见"TestBase → APITestBase → LoginTest"的深层继承链。随着业务增长，基类不断膨胀（截图、日志、HTTP、DB 等），最终变成"上帝类"。组合模式能解决这个问题。

组合优于继承
# 用组合 (Mixin / 依赖注入) 替代深层继承
class ScreenshotMixin:
    def take_screenshot(self, name):
        self.page.screenshot(path=f"screenshots/{name}.png")

class LoggerMixin:
    def log_step(self, step):
        print(f"[{self.__class__.__name__}] {step}")

class LoginTest(ScreenshotMixin, LoggerMixin):
    """平铺的 Mixin 组合，每个 Mixin 职责单一"""
    def __init__(self, page):
        self.page = page

    def test_login(self):
        self.log_step("打开登录页")
        self.page.goto("/login")
        self.page.fill("[name=user]", "admin")
        self.take_screenshot("after_login")
        self.log_step("登录完成")
💡 OOP 在测试框架中的最佳实践
  • 组合优于继承：Mixin 类比深层继承链更易维护，每个 Mixin 只做一件事。
  • 依赖注入：不要 inside class 中 import requests 并 hard-code URL，将 client 作为构造函数参数传入，便于测试时 mock。
  • 避免 God Object：若一个类同时处理 HTTP、DB、文件、认证，立即拆分。
⚠️ 避坑指南：继承陷阱与 super() 调用
  • MRO 顺序：Python 使用 C3 线性化算法决定多重继承的方法查找顺序。使用 ClassName.__mro__ 查看。
  • super() 缺失：覆盖 __init__ 时忘记调用 super().__init__() 会导致父类属性未初始化，产生难以调试的 AttributeError。
  • 测试数据污染：若父类 __init__ 中创建了可变默认参数（如 list/dict），所有子类实例会共享同一个对象 — 这是经典的 Python 陷阱。
