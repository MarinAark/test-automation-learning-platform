变量与数据类型

变量与类型
# 基本数据类型
name = "测试工程师"          # str
age = 28                     # int
salary = 15.5                # float
is_senior = True             # bool
skills = ["Python", "pytest", "Selenium"]  # list
config = {"env": "staging", "timeout": 30} # dict
tags = {"smoke", "regression"}             # set

# 类型检查
print(type(name))   # <class 'str'>
print(isinstance(age, int))  # True

# f-string 格式化
print(f"{name}，{age}岁，月薪{salary}K")

控制流

条件与循环
# if-elif-else
status_code = 404
if status_code == 200:
    result = "通过"
elif status_code in (301, 302):
    result = "重定向"
elif 400 <= status_code < 500:
    result = "客户端错误"
else:
    result = "服务端错误"

# for 循环
test_cases = ["login", "register", "checkout"]
for case in test_cases:
    print(f"执行用例: {case}")

# 列表推导式
statuses = [200, 404, 500, 200, 302]
errors = [s for s in statuses if s >= 400]
print(f"失败状态码: {errors}")  # [404, 500]

# 字典推导式
case_results = {"login": True, "register": False, "checkout": True}
passed = {k: v for k, v in case_results.items() if v}
print(f"通过的用例: {list(passed.keys())}")

常用内置函数

内置函数
# enumerate — 带索引遍历
cases = ["TC001", "TC002", "TC003"]
for i, case in enumerate(cases, 1):
    print(f"第{i}个用例: {case}")

# zip — 并行遍历
names = ["登录", "注册", "下单"]
ids = ["TC001", "TC002", "TC003"]
for name, tid in zip(names, ids):
    print(f"{tid}: {name}")

# map / filter
results = [200, 404, 500, 200]
all_ok = all(r == 200 for r in results)  # False
any_error = any(r >= 400 for r in results)  # True

# sorted — 自定义排序
bugs = [{"severity": "P2"}, {"severity": "P0"}, {"severity": "P1"}]
sorted_bugs = sorted(bugs, key=lambda b: b["severity"])
print(sorted_bugs)
