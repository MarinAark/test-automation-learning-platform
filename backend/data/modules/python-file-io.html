import json

# 读取测试用例
def load_test_cases(filepath):
    with open(filepath, "r", encoding="utf-8") as f:
        return json.load(f)

# 写入测试结果
def save_results(filepath, results):
    with open(filepath, "w", encoding="utf-8") as f:
        json.dump(results, f, ensure_ascii=False, indent=2)

CSV 数据处理

CSV 操作
import csv

# 读取 CSV 测试数据
def load_csv_cases(filepath):
    with open(filepath, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        return list(reader)

# 写入 CSV 报告
def write_csv_report(filepath, results):
    with open(filepath, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=["case", "status", "duration"])
        writer.writeheader()
        writer.writerows(results)

# 使用
results = [
    {"case": "登录", "status": "PASS", "duration": 0.5},
    {"case": "注册", "status": "PASS", "duration": 0.3},
    {"case": "下单", "status": "FAIL", "duration": 2.1},
]
write_csv_report("test_report.csv", results)

YAML 配置

YAML 配置
# 需要 pip install pyyaml
import yaml

# config/settings.yaml 内容:
'''
staging:
  base_url: https://staging.example.com
  timeout: 30
  retry: 2
production:
  base_url: https://api.example.com
  timeout: 15
  retry: 0
'''

def load_config(env="staging"):
    """加载多环境配置"""
    with open("config/settings.yaml", "r", encoding="utf-8") as f:
        all_config = yaml.safe_load(f)
    return all_config.get(env, {})

config = load_config("staging")
print(f"测试环境: {config['base_url']}")
print(f"超时设置: {config['timeout']}s")
