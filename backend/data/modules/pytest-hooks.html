def pytest_collection_modifyitems(config, items):
    '''Auto-tag tests based on path'''
    for item in items:
        if "api" in item.nodeid: item.add_marker(pytest.mark.api)
        elif "ui" in item.nodeid: item.add_marker(pytest.mark.ui)
