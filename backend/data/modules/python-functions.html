函数定义与参数

函数参数
def run_test(name, env="staging", *tags, **options):
    """执行测试用例"""
    timeout = options.get("timeout", 30)
    retry = options.get("retry", 0)
    print(f"[{env}] {name} tags={tags} timeout={timeout}s retry={retry}")

# 调用方式
run_test("登录用例")
run_test("下单用例", "production", "smoke", "p0", timeout=60, retry=2)

# 解包参数
config = {"env": "staging", "timeout": 45, "retry": 1}
run_test("注册用例", **config)

Lambda 与高阶函数

lambda 示例
# 测试数据排序
test_results = [
    {"case": "登录", "duration": 0.5},
    {"case": "下单", "duration": 2.1},
    {"case": "注册", "duration": 0.3},
]
# 按耗时排序
sorted_by_time = sorted(test_results, key=lambda r: r["duration"])
# 找出耗时超过1秒的用例
slow_cases = list(filter(lambda r: r["duration"] > 1, test_results))

print(f"最慢用例: {sorted_by_time[-1]['case']}")
print(f"慢查询: {[c['case'] for c in slow_cases]}")

装饰器基础

装饰器
import time
from functools import wraps

def timeit(func):
    """计时装饰器 — 自动记录函数执行时间"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} 耗时: {elapsed:.3f}s")
        return result
    return wrapper

def retry(max_attempts=3):
    """重试装饰器 — 失败自动重试"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts:
                        raise
                    print(f"重试 {attempt}/{max_attempts}: {e}")
                    time.sleep(1)
        return wrapper
    return decorator

# 使用
@timeit
@retry(max_attempts=2)
def unstable_api_call():
    # 模拟不稳定接口
    import random
    if random.random() > 0.5:
        raise ConnectionError("网络超时")
    return "success"

生成器

生成器
# 分批读取大量测试数据（节省内存）
def read_test_data(filename, batch_size=100):
    """逐批读取测试数据"""
    batch = []
    for line in open(filename, encoding="utf-8"):
        batch.append(line.strip())
        if len(batch) >= batch_size:
            yield batch
            batch = []
    if batch:
        yield batch

# 使用
for batch in read_test_data("large_test_cases.csv"):
    print(f"处理 {len(batch)} 条用例")

实战场景：装饰器 vs 显式重试 — 何时用哪个？

自动化测试中有两种常见的重试策略。装饰器适合"无侵入"地给已有函数加能力；但当重试参数与业务逻辑耦合时，显式控制更清晰。

💡 选择指南
  • 用装饰器：当你希望对所有函数统一添加重试、计时、日志等横切关注点时。例如：整个项目的 HTTP 调用统一加持 @auto_retry。
  • 用显式重试：当重试参数（次数、间隔、退避策略）依赖于调用方的上下文时。例如：不同接口需要不同的重试策略，使用 for retry in range(n): 直接控制更灵活。
  • 典型反模式：使用同一个全局重试装饰器处理所有异常（包括断言错误），导致测试失败后被反复重试，浪费 CI 资源且日志混乱。应在装饰器中加 retry_on 白名单参数。
⚠️ 避坑指南：@wraps 丢失与装饰器堆叠顺序
  • 丢失元数据：若装饰器未使用 @wraps(func)，被装饰函数的 __name__ 会变为 wrapper。pytest 的 -k 按名称筛选用例时会失败。
  • 堆叠顺序：@timeit 和 @retry 的上下顺序决定执行流程。装饰器从下往上执行（最靠近函数的最先包裹），理解这一点对正确组合装饰器至关重要。
  • 参数化装饰器：带参数的装饰器实际上是返回装饰器的函数，需要额外一层嵌套，初学者容易遗漏最外层。
